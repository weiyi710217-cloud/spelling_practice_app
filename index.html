} } } if (items.length > 0) return items; // 2. 備援：若模型仍回傳 JSON，使用容錯正規表達式逐筆提取，不怕中間斷掉或雙引號錯誤 try { const cleanJson = rawText.replace(/^```json\s*/i, '').replace(/^```\s*/i, '').replace(/```\s*$/i, '').trim();
